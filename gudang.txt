/* ============================================================
   HELPER
============================================================ */

function esc($value)
{
    return htmlspecialchars(
        (string)($value ?? ''),
        ENT_QUOTES,
        'UTF-8'
    );
}


function dbesc($value)
{
    global $conn;

    return mysqli_real_escape_string(
        $conn,
        trim((string)($value ?? ''))
    );
}

/* ============================================================
   FUNGSI CEK KOLOM
============================================================ */

function getTableColumns($table)
{
    global $conn;

    $columns = [];

    $safeTable = str_replace('`', '``', $table);

    $result = mysqli_query(
        $conn,
        "SHOW COLUMNS FROM `{$safeTable}`"
    );

    if (!$result) {

        showError(
            'Database Error',
            'Gagal membaca struktur tabel ' .
            $table .
            ': ' .
            mysqli_error($conn)
        );
    }

    while ($row = mysqli_fetch_assoc($result)) {
        $columns[] = $row['Field'];
    }

    return $columns;
}


/* ============================================================
   FUNGSI CEK TABEL
============================================================ */

function tableExists($table)
{
    global $conn;

    $table = mysqli_real_escape_string(
        $conn,
        $table
    );

    $result = mysqli_query(
        $conn,
        "SHOW TABLES LIKE '{$table}'"
    );

    return $result &&
           mysqli_num_rows($result) > 0;
}

/* ============================================================
   PROSES RESTORE
============================================================ */

if (
    $_SERVER['REQUEST_METHOD'] === 'POST' &&
    isset($_POST['restore_gudang'])
) {

    $idGudang = intval(
        $_POST['id'] ?? 0
    );


    /* --------------------------------------------------------
       VALIDASI ID GUDANG
    -------------------------------------------------------- */

    if ($idGudang <= 0) {

        showError(
            'Restore Gagal',
            'ID data gudang tidak valid.'
        );
    }


    /* --------------------------------------------------------
       MULAI TRANSACTION
    -------------------------------------------------------- */

    mysqli_begin_transaction($conn);


    try {

        /* ----------------------------------------------------
           AMBIL DATA DARI GUDANG
        ---------------------------------------------------- */

        $stmt = mysqli_prepare(
            $conn,
            "
            SELECT *
            FROM gudang_kartika2
            WHERE id = ?
            LIMIT 1
            "
        );


        if (!$stmt) {

            throw new Exception(
                'Gagal mengambil data gudang: ' .
                mysqli_error($conn)
            );
        }


        mysqli_stmt_bind_param(
            $stmt,
            "i",
            $idGudang
        );


        if (
            !mysqli_stmt_execute($stmt)
        ) {

            $error = mysqli_stmt_error($stmt);

            mysqli_stmt_close($stmt);

            throw new Exception(
                'Gagal mengambil data gudang: ' .
                $error
            );
        }


        $result =
            mysqli_stmt_get_result($stmt);


        $dataGudang =
            mysqli_fetch_assoc($result);


        mysqli_stmt_close($stmt);


        /* ----------------------------------------------------
           DATA TIDAK DITEMUKAN
        ---------------------------------------------------- */

        if (!$dataGudang) {

            throw new Exception(
                'Data gudang tidak ditemukan.'
            );
        }


        /* ----------------------------------------------------
           AMBIL ID ASAL
           
           INI ADALAH ID ASLI DI
           inventaris_kartika2
        ---------------------------------------------------- */

        $idAsal = intval(
            $dataGudang['id_asal'] ?? 0
        );


        if ($idAsal <= 0) {

            throw new Exception(
                'Data gudang tidak mempunyai id_asal yang valid.'
            );
        }


        /* ----------------------------------------------------
           CEK TABEL INVENTARIS
        ---------------------------------------------------- */

        if (
            !tableExists(
                'inventaris_kartika2'
            )
        ) {

            throw new Exception(
                'Tabel inventaris_kartika2 tidak ditemukan.'
            );
        }


        /* ----------------------------------------------------
           AMBIL STRUKTUR KOLOM
        ---------------------------------------------------- */

        $columnsGudang =
            getTableColumns(
                'gudang_kartika2'
            );


        $columnsInventaris =
            getTableColumns(
                'inventaris_kartika2'
            );


        /* ----------------------------------------------------
           CARI KOLOM YANG SAMA
        ---------------------------------------------------- */

        $commonColumns =
            array_intersect(
                $columnsGudang,
                $columnsInventaris
            );


        /*
         * Jangan mengubah:
         *
         * id
         * id_asal
         * status_gudang
         *
         * karena:
         *
         * id         = ID asli inventaris
         * id_asal    = referensi gudang
         * status_gudang hanya milik gudang
         */

        $commonColumns =
            array_values(
                array_diff(
                    $commonColumns,
                    [
                        'id',
                        'id_asal',
                        'status_gudang'
                    ]
                )
            );


        /* ====================================================
           CEK APAKAH ID ASLI MASIH ADA
        ==================================================== */

        $stmt = mysqli_prepare(
            $conn,
            "
            SELECT id
            FROM inventaris_kartika2
            WHERE id = ?
            LIMIT 1
            "
        );


        if (!$stmt) {

            throw new Exception(
                'Gagal memeriksa data inventaris: ' .
                mysqli_error($conn)
            );
        }


        mysqli_stmt_bind_param(
            $stmt,
            "i",
            $idAsal
        );


        if (
            !mysqli_stmt_execute($stmt)
        ) {

            $error = mysqli_stmt_error($stmt);

            mysqli_stmt_close($stmt);

            throw new Exception(
                'Gagal memeriksa data inventaris: ' .
                $error
            );
        }


        $result =
            mysqli_stmt_get_result($stmt);


        $sudahAda =
            mysqli_fetch_assoc($result);


        mysqli_stmt_close($stmt);


        /* ====================================================
           JIKA DATA ASLI MASIH ADA
           
           UPDATE DATA ASLI
        ==================================================== */

        if ($sudahAda) {

            /* ------------------------------------------------
               Jika tidak ada kolom yang cocok
            ------------------------------------------------ */

            if (empty($commonColumns)) {

                throw new Exception(
                    'Tidak ditemukan kolom yang cocok antara gudang_kartika2 dan inventaris_kartika2.'
                );
            }


            /* ------------------------------------------------
               BENTUK SET DINAMIS
            ------------------------------------------------ */

            $setParts = [];
            $updateValues = [];


            foreach (
                $commonColumns as $column
            ) {

                /*
                 * Jangan update kolom id.
                 */

                if ($column === 'id') {
                    continue;
                }


                $setParts[] =
                    '`' .
                    str_replace(
                        '`',
                        '``',
                        $column
                    ) .
                    '` = ?';


                $updateValues[] =
                    (string)(
                        $dataGudang[$column]
                        ?? ''
                    );
            }


            if (empty($setParts)) {

                throw new Exception(
                    'Tidak ada kolom yang dapat diperbarui pada inventaris_kartika2.'
                );
            }


            /* ------------------------------------------------
               UPDATE DATA ASLI
            ------------------------------------------------ */

            $updateSql =

                "UPDATE `inventaris_kartika2`
                 SET " .
                 implode(
                     ', ',
                     $setParts
                 ) .
                 "
                 WHERE `id` = ?
                 LIMIT 1";


            $stmt =
                mysqli_prepare(
                    $conn,
                    $updateSql
                );


            if (!$stmt) {

                throw new Exception(
                    'Gagal menyiapkan UPDATE inventaris: ' .
                    mysqli_error($conn)
                );
            }


            /*
             * Tambahkan id asli sebagai parameter terakhir.
             */

            $updateValues[] =
                $idAsal;


            /*
             * Semua kolom data dianggap string,
             * ID terakhir integer.
             */

            $types =
                str_repeat(
                    's',
                    count($updateValues) - 1
                ) . 'i';


            $bindParams = [];

            $bindParams[] =
                $stmt;

            $bindParams[] =
                $types;


            foreach (
                $updateValues
                as $key => $value
            ) {

                $bindParams[] =
                    &$updateValues[$key];
            }


            call_user_func_array(
                'mysqli_stmt_bind_param',
                $bindParams
            );


            if (
                !mysqli_stmt_execute($stmt)
            ) {

                $error =
                    mysqli_stmt_error($stmt);

                mysqli_stmt_close($stmt);

                throw new Exception(
                    'Gagal mengembalikan data ke inventaris_kartika2: ' .
                    $error
                );
            }


            mysqli_stmt_close($stmt);


        }

        /* ====================================================
           JIKA DATA ASLI SUDAH TIDAK ADA
           
           INSERT KEMBALI DENGAN ID ASLI
        ==================================================== */

        else {

            /*
             * Kolom INSERT:
             *
             * id = id_asal
             */

            $insertColumns = [
                'id'
            ];


            $insertValues = [
                $idAsal
            ];


            foreach (
                $commonColumns
                as $column
            ) {

                $insertColumns[] =
                    $column;


                $insertValues[] =
                    $dataGudang[$column]
                    ?? null;
            }


            /* ------------------------------------------------
               QUOTE NAMA KOLOM
            ------------------------------------------------ */

            $quotedColumns = [];


            foreach (
                $insertColumns
                as $column
            ) {

                $quotedColumns[] =
                    '`' .
                    str_replace(
                        '`',
                        '``',
                        $column
                    ) .
                    '`';
            }


            /* ------------------------------------------------
               PLACEHOLDER
            ------------------------------------------------ */

            $placeholders =
                implode(
                    ', ',
                    array_fill(
                        0,
                        count($insertValues),
                        '?'
                    )
                );


            /* ------------------------------------------------
               SQL INSERT
            ------------------------------------------------ */

            $insertSql =

                "INSERT INTO `inventaris_kartika2`
                (" .
                implode(
                    ', ',
                    $quotedColumns
                ) .
                ")
                VALUES (" .
                $placeholders .
                ")";


            $stmt =
                mysqli_prepare(
                    $conn,
                    $insertSql
                );


            if (!$stmt) {

                throw new Exception(
                    'Gagal menyiapkan INSERT inventaris: ' .
                    mysqli_error($conn)
                );
            }


            /* ------------------------------------------------
               SEMUA VALUE STRING
            ------------------------------------------------ */

            $types =
                str_repeat(
                    's',
                    count($insertValues)
                );


            $bindParams = [];

            $bindParams[] =
                $stmt;

            $bindParams[] =
                $types;


            foreach (
                $insertValues
                as $key => $value
            ) {

                /*
                 * NULL tetap dikirim sebagai string kosong
                 * untuk kompatibilitas dengan struktur tabel.
                 */

                if ($value === null) {
                    $insertValues[$key] = '';
                }


                $bindParams[] =
                    &$insertValues[$key];
            }


            call_user_func_array(
                'mysqli_stmt_bind_param',
                $bindParams
            );


            /* ------------------------------------------------
               EXECUTE INSERT
            ------------------------------------------------ */

            if (
                !mysqli_stmt_execute($stmt)
            ) {

                $error =
                    mysqli_stmt_error($stmt);

                mysqli_stmt_close($stmt);

                throw new Exception(
                    'Gagal membuat kembali data di inventaris_kartika2: ' .
                    $error
                );
            }


            mysqli_stmt_close($stmt);
        }


        /* ====================================================
           SETELAH DATA BERHASIL DIKEMBALIKAN
           
           BARU HAPUS DATA DARI GUDANG
        ==================================================== */

        $stmt = mysqli_prepare(
            $conn,
            "
            DELETE FROM gudang_kartika2
            WHERE id = ?
            LIMIT 1
            "
        );


        if (!$stmt) {

            throw new Exception(
                'Gagal menyiapkan penghapusan data gudang: ' .
                mysqli_error($conn)
            );
        }


        mysqli_stmt_bind_param(
            $stmt,
            "i",
            $idGudang
        );


        if (
            !mysqli_stmt_execute($stmt)
        ) {

            $error =
                mysqli_stmt_error($stmt);

            mysqli_stmt_close($stmt);

            throw new Exception(
                'Gagal menghapus data dari gudang: ' .
                $error
            );
        }


        /*
         * Pastikan memang ada data yang dihapus.
         */

        $affectedRows =
            mysqli_stmt_affected_rows($stmt);


        mysqli_stmt_close($stmt);


        if ($affectedRows <= 0) {

            throw new Exception(
                'Data gudang tidak berhasil dihapus setelah proses restore.'
            );
        }


        /* ====================================================
           COMMIT
        ==================================================== */

        mysqli_commit($conn);


        /* ====================================================
           KEMBALI KE GUDANG
        ==================================================== */

        header(
            'Location: gudang.php?status=restore_sukses'
        );

        exit;


    } catch (Throwable $e) {

        /*
         * Jika salah satu proses gagal,
         * seluruh perubahan dibatalkan.
         */

        mysqli_rollback($conn);


        showError(
            'Restore ke Inventaris Gagal',
            $e->getMessage()
        );
    }
}

            'id_personil_edp'

        ];


        foreach (
            $pcFields
            as $field
        ) {

            $payload[$field] = '';
        }
    }


    /*
     * PRINTER 2 SAJA
     */

    elseif (
        $tipeKirim === 'printer2'
    ) {

        $pcFields = [

            'nama_pc',
            'nama_user',

            'manufactur',
            'processor_type',
            'windows_version',

            'ram_size',
            'ram_hardisk',

            'monitor_model',
            'monitor_size',
            'monitor_vga',

            'printer_1',

            'tahun_pengadaan',
            'id_personil_edp'

        ];


        foreach (
            $pcFields
            as $field
        ) {

            $payload[$field] = '';
        }
    }


    /*
     * SEMUA
     *
     * PC + Printer yang tersedia
     * ikut dikirim.
     */


    /* ========================================================
       LOKASI TUJUAN
    ======================================================== */

    $payload['gedung'] =
        $gedungTujuan;

    $payload['lantai'] =
        $lantaiTujuan;

    $payload['ruangan'] =
        $ruanganTujuan;


    /*
     * Nama user tujuan
     */

    if (
        $userTujuan !== ''
    ) {

        $payload['nama_user'] =
            $userTujuan;
    }


    /* ========================================================
       JENIS BARANG
    ======================================================== */

    $jenisBarang = 'Lainnya';


    if (
        $tipeKirim === 'pc'
    ) {

        $jenisBarang = 'PC';

    } elseif (
        $tipeKirim === 'printer1'
    ) {

        $jenisBarang = 'Printer 1';

    } elseif (
        $tipeKirim === 'printer2'
    ) {

        $jenisBarang = 'Printer 2';

    } elseif (
        $tipeKirim === 'semua'
    ) {

        if (
            $hasPC &&
            ($hasPrinter1 || $hasPrinter2)
        ) {

            $jenisBarang =
                'PC + Printer';

        } elseif ($hasPC) {

            $jenisBarang =
                'PC';

        } elseif ($hasPrinter1) {

            $jenisBarang =
                'Printer 1';

        } elseif ($hasPrinter2) {

            $jenisBarang =
                'Printer 2';
        }
    }


    /* ========================================================
       TAMBAHAN METADATA
    ======================================================== */

    $destinationColumns =
        getTableColumns(
            $destinationTable
        );


    if (
        in_array(
            'status',
            $destinationColumns,
            true
        )
    ) {

        $payload['status'] =
            'Aktif';
    }


    if (
        in_array(
            'status_barang',
            $destinationColumns,
            true
        )
    ) {

        $payload['status_barang'] =
            'Aktif';
    }


    if (
        in_array(
            'jenis_barang',
            $destinationColumns,
            true
        )
    ) {

        $payload['jenis_barang'] =
            $jenisBarang;
    }


    if (
        in_array(
            'keterangan',
            $destinationColumns,
            true
        )
    ) {

        $payload['keterangan'] =
            $catatanKirim;
    }


    if (
        in_array(
            'catatan',
            $destinationColumns,
            true
        )
    ) {

        $payload['catatan'] =
            $catatanKirim;
    }


    $now =
        date('Y-m-d H:i:s');


    if (
        in_array(
            'tanggal_masuk',
            $destinationColumns,
            true
        )
    ) {

        $payload['tanggal_masuk'] =
            $now;
    }


    if (
        in_array(
            'tanggal',
            $destinationColumns,
            true
        )
    ) {

        $payload['tanggal'] =
            $now;

    if (
        in_array(
            'created_at',
            $destinationColumns,

        )
    ) {

        $payload['created_at'] =
            $now;
    }


    /* ========================================================
       HANYA KOLOM YANG ADA DI TABEL TUJUAN
    ======================================================== */

    $insertColumns = [];
    $insertValues = [];


    foreach (
        $payload
        as $field => $value
    ) {

        if (
            $field === 'id'
        ) {
            continue;
        }


        if (
            in_array(
                $field,
                $destinationColumns,

                str_replace(
                    '`',
                    '``',

            implode(
                ', ',
                array_fill(
                    0,
                    count($insertColumns),
                    '?'
                )

            str_replace(
                '`',
                '``',
                $destinationTable
            ) .
            "` (" .
            implode(
                ', ',
                $quotedColumns
            ) .
            ") VALUES (" .
            $placeholders .
            ")";


        $stmt =
            mysqli_prepare(
                $conn,
                $insertSql
            );


        if (!$stmt) {

            throw new Exception(
                'Gagal menyiapkan INSERT: ' .
                mysqli_error($conn)
            );
        }


        /* ----------------------------------------------------
           BIND PARAMETER DINAMIS
        ---------------------------------------------------- */

        $types =
            str_repeat(
                's',
                count($insertValues)
            );


        $bindParams = [];

        $bindParams[] =
            $stmt;

        $bindParams[] =
            $types;


        foreach (
            $insertValues
            as $key => $value
        ) {

            $bindParams[] =
                &$insertValues[$key];
        }


        call_user_func_array(
            'mysqli_stmt_bind_param',
            $bindParams
        );


        /* ----------------------------------------------------
           EXECUTE INSERT
        ---------------------------------------------------- */

        if (
            !mysqli_stmt_execute($stmt)
        ) {

            $error =
                mysqli_stmt_error($stmt);

            mysqli_stmt_close($stmt);

            throw new Exception(
                'Gagal memasukkan data ke ' .
                $destinationTable .
                ': ' .
                $error
            );
        }


        mysqli_stmt_close($stmt);


        /* ====================================================
           UPDATE / DELETE GUDANG
        ==================================================== */

        /*
         * PC SAJA
         *
         * PC dihapus dari gudang,
         * printer tetap.
         */

        if (
            $tipeKirim === 'pc'
        ) {

            $sqlUpdate = "

                UPDATE gudang_kartika2

                SET

                    nama_pc = '',
                    nama_user = '',
                    manufactur = '',
                    processor_type = '',
                    windows_version = '',
                    ram_size = '',
                    ram_hardisk = '',
                    monitor_model = '',
                    monitor_size = '',
                    monitor_vga = '',
                    tahun_pengadaan = '',
                    id_personil_edp = ''

                WHERE id = ?

                LIMIT 1
            ";
        }


        /*
         * PRINTER 1 SAJA
         *
         * PC tetap.
         * Printer 2 tetap.
         */

        elseif (
            $tipeKirim === 'printer1'
        ) {

            $sqlUpdate = "

                UPDATE gudang_kartika2

                SET

                    printer_1 = ''

                WHERE id = ?

                LIMIT 1
            ";
        }


        /*
         * PRINTER 2 SAJA
         */

        elseif (
            $tipeKirim === 'printer2'
        ) {

            $sqlUpdate = "

                UPDATE gudang_kartika2

                SET

                    printer_2 = ''

                WHERE id = ?

                LIMIT 1
            ";
        }


        /*
         * SEMUA
         *
         * Seluruh isi record dipindahkan.
         */

        else {

            $sqlUpdate = "

                DELETE FROM gudang_kartika2

                WHERE id = ?

                LIMIT 1
            ";
        }


        /* ----------------------------------------------------
           EXECUTE UPDATE GUDANG
        ---------------------------------------------------- */

        $stmt =
            mysqli_prepare(
                $conn,
                $sqlUpdate
            );


        if (!$stmt) {

            throw new Exception(
                'Gagal menyiapkan perubahan gudang: ' .
                mysqli_error($conn)
            );
        }


        mysqli_stmt_bind_param(
            $stmt,
            'i',
            $idGudang
        );


        if (
            !mysqli_stmt_execute($stmt)
        ) {

            $error =
                mysqli_stmt_error($stmt);

            mysqli_stmt_close($stmt);

            throw new Exception(
                'Gagal memperbarui gudang: ' .
                $error
            );
        }


        mysqli_stmt_close($stmt);


        /* ----------------------------------------------------
           COMMIT
        ---------------------------------------------------- */

        mysqli_commit(
            $conn
        );


        header(
            'Location: gudang.php?status=kirim_sukses&tujuan=' .
            urlencode($destinationTable)
        );

        exit;


    } catch (Throwable $e) {

        mysqli_rollback(
            $conn
        );


        showError(
            'Kirim ke Tujuan Gagal',
            $e->getMessage()
        );
    }
}

<script>


/* =========================================================
   MODAL
========================================================= */

function openModal(id)
{
    document.getElementById(id).style.display =
        'block';
}


function closeModal(id)
{
    document.getElementById(id).style.display =
        'none';
}


/* =========================================================
   CLOSE KLIK LUAR
========================================================= */

window.addEventListener(
    'click',
    function(event)
    {

        document.querySelectorAll(
            '.modal'
        ).forEach(
            function(modal)
            {

                if (
                    event.target === modal
                ) {

                    modal.style.display =
                        'none';
                }

            }
        );

    }
);


/* =========================================================
   KIRIM MODAL
========================================================= */

function openKirimModal(row)
{

    document.getElementById(
        'kirim_id'
    ).value =
        row.id || '';


    /*
     * Reset pilihan.
     */

    document.querySelectorAll(
        '#modalKirim input[type="radio"]'
    ).forEach(
        function(el)
        {
            el.checked = false;
            el.disabled = false;
        }
    );


    /*
     * Tentukan komponen yang tersedia.
     */

    var hasPC =
        String(
            row.nama_pc || ''
        ).trim() !== '';


    var hasPrinter1 =
        String(
            row.printer_1 || ''
        ).trim() !== '';


    var hasPrinter2 =
        String(
            row.printer_2 || ''
        ).trim() !== '';


    document.getElementById(
        'kirim_pc'
    ).disabled =
        !hasPC;


    document.getElementById(
        'kirim_printer1'
    ).disabled =
        !hasPrinter1;


    document.getElementById(
        'kirim_printer2'
    ).disabled =
        !hasPrinter2;


    document.getElementById(
        'kirim_semua'
    ).disabled =
        !(
            hasPC ||
            hasPrinter1 ||
            hasPrinter2
        );


    /*
     * Preview.
     */

    var html = '';

    html +=
        '<b>ID Gudang:</b> ' +
        escapeHtml(
            row.id || ''
        ) +
        '<br>';


    html +=
        '<b>PC:</b> ' +
        escapeHtml(
            row.nama_pc || '-'
        ) +
        '<br>';


    html +=
        '<b>Printer 1:</b> ' +
        escapeHtml(
            row.printer_1 || '-'
        ) +
        '<br>';


    html +=
        '<b>Printer 2:</b> ' +
        escapeHtml(
            row.printer_2 || '-'
        );


    document.getElementById(
        'previewKirim'
    ).innerHTML =
        html;


    openModal(
        'modalKirim'
    );
}


/* =========================================================
   VALIDASI KIRIM
========================================================= */

function validateKirim()
{

    var tipe =
        document.querySelector(
            'input[name="tipe_kirim"]:checked'
        );


    var tujuan =
        document.querySelector(
            'input[name="tujuan_kirim"]:checked'
        );


    if (!tipe) {

        alert(
            'Silakan pilih barang yang akan dikirim.'
        );

        return false;
    }


    if (!tujuan) {

        alert(
            'Silakan pilih tujuan pengiriman.'
        );

        return false;
    }


    var tujuanText =
        tujuan
            .nextElementSibling
            .innerText
            .trim();


    var tipeText =
        tipe
            .nextElementSibling
            .innerText
            .trim();


    return confirm(
        'Kirim ' +
        tipeText +
        ' ke ' +
        tujuanText +
        '?'
    );
}

    ).value =
        status;


    openModal(
        'modalStatus'
    );
}

function showDetail(row)
{

    var fields = [

        ['ID', row.id],

        ['ID Asal', row.id_asal],

        ['Jenis Barang', row.jenis_barang],

        ['Nama PC', row.nama_pc],

        ['Nama User', row.nama_user],

        ['Manufactur', row.manufactur],

        ['Processor', row.processor_type],

        ['Windows', row.windows_version],

        ['RAM', row.ram_size],

        ['Harddisk', row.ram_hardisk],

        ['Monitor', row.monitor_model],

        ['Ukuran Monitor', row.monitor_size],

        ['VGA Monitor', row.monitor_vga],

        ['Printer 1', row.printer_1],

        ['Printer 2', row.printer_2],

        ['Gedung', row.gedung],

        ['Lantai', row.lantai],

        ['Ruangan', row.ruangan],

        ['Tahun Pengadaan', row.tahun_pengadaan],

        ['Status Gudang', row.status_gudang],

        ['Alasan Ganti', row.alasan_ganti]

    ];


    var html = '';


    fields.forEach(
        function(item)
        {

            html +=

                '<div class="detail-item">' +

                    '<div class="detail-label">' +

                        escapeHtml(
                            item[0]
                        ) +

                    '</div>' +

                    '<div class="detail-value">' +

                        escapeHtml(
                            item[1] || '-'
                        ) +

                    '</div>' +

                '</div>';

        }
    );


    document.getElementById(
        'detailContent'
    ).innerHTML =
        html;


    openModal(
        'modalDetail'
    );
}


/* =========================================================
   ESCAPE HTML JAVASCRIPT
========================================================= */

function escapeHtml(value)
{

    var div =
        document.createElement(
            'div'
        );


    div.textContent =
        value == null
            ? ''
            : String(value);


    return div.innerHTML;
}
