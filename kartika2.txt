    $tipe_valid = ['pc', 'printer1', 'printer2', 'semua'];

    if (!in_array($tipe_ganti, $tipe_valid, true)) {
        die('Jenis barang yang dipilih tidak valid.');
    }

    if (empty($alasan)) {
        die('Alasan penggantian wajib diisi.');
    }

    // Ambil data inventaris
    $query_data = mysqli_query(
        $conn,
        "SELECT * FROM inventaris_kartika2 WHERE id = $id LIMIT 1"
    );

    $data = mysqli_fetch_assoc($query_data);

        die('Data inventaris tidak ditemukan.');
    }

    $gedung          = mysqli_real_escape_string($conn, $data['gedung'] ?? '');
    $lantai          = mysqli_real_escape_string($conn, $data['lantai'] ?? '');

            $windows_version = mysqli_real_escape_string($conn, $data['windows_version'] ?? '');
            $ram_size        = mysqli_real_escape_string($conn, $data['ram_size'] ?? '');
            $ram_hardisk     = mysqli_real_escape_string($conn, $data['ram_hardisk'] ?? '');
            $monitor_model   = mysqli_real_escape_string($conn, $data['monitor_model'] ?? '');
            $monitor_size    = mysqli_real_escape_string($conn, $data['monitor_size'] ?? '');
            $monitor_vga     = mysqli_real_escape_string($conn, $data['monitor_vga'] ?? '');

            // Printer sengaja dikosongkan karena yang dipindahkan hanya PC
            $printer_1 = '';
            $printer_2 = '';

            $sql_gudang = "
                INSERT INTO gudang_kartika2 (
                    id_asal,
                    gedung,
                    lantai,
                    ruangan,
                    nama_pc,
                    nama_user,
                    manufactur,
                    processor_type,
                    windows_version,
                    ram_size,
                    ram_hardisk,
                    monitor_model,
                    monitor_size,
                    monitor_vga,
                    printer_1,
                    printer_2,
                    tahun_pengadaan,
                    id_personil_edp,
                    alasan_ganti,
                    status_gudang
                ) VALUES (
                    $id,
                    '$gedung',
                    '$lantai',
                    '$ruangan',
                    '$nama_pc',
                    '$nama_user',
                    '$manufactur',
                    '$processor_type',
                    '$windows_version',
                    '$ram_size',
                    '$ram_hardisk',
                    '$monitor_model',
                    '$monitor_size',
                    '$monitor_vga',
                    '',
                    '',
                    '$tahun_pengadaan',
                    '$id_personil_edp',
                    '$alasan',
                    '$status_gudang'

                UPDATE inventaris_kartika2 SET
                    nama_pc = '',
                    nama_user = '',
                    manufactur = '',
                    processor_type = '',
                    windows_version = '',
                    ram_size = '',
                    ram_hardisk = '',
                    monitor_model = '',
                    monitor_size = '',
                    monitor_vga = '',
                    tahun_pengadaan = '',
                    id_personil_edp = ''
                WHERE id = $id
            ";

            if (!mysqli_query($conn, $sql_update)) {
                throw new Exception('Gagal menghapus PC dari inventaris: ' . mysqli_error($conn));
            }
        }


        // ==========================================================
        // 2. PINDAH PRINTER 1 SAJA
        // ==========================================================
        elseif ($tipe_ganti === 'printer1') {

            // Pastikan Printer 1 memang ada
            if (trim($data['printer_1'] ?? '') === '') {
                throw new Exception('Printer 1 kosong. Tidak ada Printer 1 yang dapat dipindahkan.');
            }

            $printer_1 = mysqli_real_escape_string($conn, $data['printer_1']);

            /*
            |--------------------------------------------------------------------------
            | Hanya printer_1 yang diisi.
            | PC, monitor dan printer_2 dibuat kosong di gudang.
            |--------------------------------------------------------------------------
            */

            $sql_gudang = "
                INSERT INTO gudang_kartika2 (
                    id_asal,
                    gedung,
                    lantai,
                    ruangan,
                    nama_pc,
                    nama_user,
                    manufactur,
                    processor_type,
                    windows_version,
                    ram_size,
                    ram_hardisk,
                    monitor_model,
                    monitor_size,
                    monitor_vga,
                    printer_1,
                    printer_2,
                    tahun_pengadaan,
                    id_personil_edp,
                    alasan_ganti,
                    status_gudang
                ) VALUES (
                    $id,
                    '$gedung',
                    '$lantai',
                    '$ruangan',
                    '',
                    '',
                    '',
                    '',
                    '',
                    '',
                    '',
                    '',
                    '',
                    '',
                    '$printer_1',
                    '',
                    '',
                    '',
                    '$alasan',
                    '$status_gudang'

                throw new Exception('Gagal memindahkan Printer 1 ke gudang: ' . mysqli_error($conn));

                UPDATE inventaris_kartika2
                SET printer_1 = ''
                WHERE id = $id
            ";

            if (!mysqli_query($conn, $sql_update)) {
                throw new Exception('Gagal menghapus Printer 1 dari inventaris: ' . mysqli_error($conn));
            }
        }


        // ==========================================================
        // 3. PINDAH PRINTER 2 SAJA
        // ==========================================================
        elseif ($tipe_ganti === 'printer2') {

            // Pastikan Printer 2 memang ada
            if (trim($data['printer_2'] ?? '') === '') {
                throw new Exception('Printer 2 kosong. Tidak ada Printer 2 yang dapat dipindahkan.');
            }

            $printer_2 = mysqli_real_escape_string($conn, $data['printer_2']);

            $sql_gudang = "
                INSERT INTO gudang_kartika2 (
                    id_asal,
                    gedung,
                    lantai,
                    ruangan,
                    nama_pc,
                    nama_user,
                    manufactur,
                    processor_type,
                    windows_version,
                    ram_size,

                    status_gudang

                    $id,

                    '',
                    '',

                throw new Exception('Gagal memindahkan Printer 2 ke gudang: ' . mysqli_error($conn));
            }

            // Hanya Printer 2 yang dihapus
            $sql_update = "
                UPDATE inventaris_kartika2
                SET printer_2 = ''
                WHERE id = $id
            ";

            if (!mysqli_query($conn, $sql_update)) {
                throw new Exception('Gagal menghapus Printer 2 dari inventaris: ' . mysqli_error($conn));
            }
        }


        // ==========================================================
        // 4. PINDAH SEMUA
        // ==========================================================
        elseif ($tipe_ganti === 'semua') {

            /*
            |--------------------------------------------------------------------------
            | Semua barang dipindahkan.
            | Jadi seluruh data asli dimasukkan ke gudang.
            |--------------------------------------------------------------------------
            */

            $fields = [
                'nama_pc',
                'nama_user',
                'manufactur',
                'processor_type',
                'windows_version',
                'ram_size',
                'ram_hardisk',
                'monitor_model',
                'monitor_size',
                'monitor_vga',
                'printer_1',
                'printer_2'
            ];

            $escaped = [];

            foreach ($fields as $field) {
                $escaped[$field] = mysqli_real_escape_string(
                    $conn,
                    $data[$field] ?? ''
                );
            }

            $sql_gudang = "
                INSERT INTO gudang_kartika2 (
                    id_asal,
                    gedung,
                    lantai,
                    ruangan,
                    nama_pc,
                    nama_user,
                    manufactur,
                    processor_type,
                    windows_version,
                    ram_size,
                    ram_hardisk,
                    monitor_model,
                    monitor_size,
                    monitor_vga,
                    printer_1,
                    printer_2,
                    tahun_pengadaan,
                    id_personil_edp,
                    alasan_ganti,
                    status_gudang
                ) VALUES (
                    $id,
                    '$gedung',
                    '$lantai',
                    '$ruangan',
                    '{$escaped['nama_pc']}',
                    '{$escaped['nama_user']}',
                    '{$escaped['manufactur']}',
                    '{$escaped['processor_type']}',
                    '{$escaped['windows_version']}',
                    '{$escaped['ram_size']}',
                    '{$escaped['ram_hardisk']}',
                    '{$escaped['monitor_model']}',
                    '{$escaped['monitor_size']}',
                    '{$escaped['monitor_vga']}',
                    '{$escaped['printer_1']}',
                    '{$escaped['printer_2']}',
                    '$tahun_pengadaan',
                    '$id_personil_edp',
                    '$alasan',
                    '$status_gudang'
                )
            ";

            if (!mysqli_query($conn, $sql_gudang)) {
                throw new Exception('Gagal memindahkan semua barang ke gudang: ' . mysqli_error($conn));
            }

            // Hapus seluruh record inventaris
            $sql_delete = "
                DELETE FROM inventaris_kartika2
                WHERE id = $id
            ";

            if (!mysqli_query($conn, $sql_delete)) {
                throw new Exception('Gagal menghapus data inventaris: ' . mysqli_error($conn));
            }
        }


        // ==========================================================
        // COMMIT
        // ==========================================================

        mysqli_commit($conn);

        header('Location: kartika2.php?success=ganti');
        exit;

    } catch (Exception $e) {

        // Batalkan semua perubahan jika ada kesalahan
        mysqli_rollback($conn);

        die(
            '<div style="
                font-family:Arial;
                padding:30px;
                background:#fee2e2;
                color:#991b1b;
                max-width:700px;
                margin:50px auto;
                border-radius:12px;
            ">
                <h3>Gagal Memindahkan Barang</h3>
                <p>' . htmlspecialchars($e->getMessage()) . '</p>
                <a href="kartika2.php">Kembali ke Kartika 2</a>
            </div>'
        );
    }
}
// ===== PROSES HAPUS DATA =====

    mysqli_query($conn, "DELETE FROM inventaris_kartika2 WHERE id=$id");
    header('Location: kartika2.php'); exit;
}
